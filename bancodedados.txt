CREATE TABLE usuario(
cpf_cnpj VARCHAR(20) PRIMARY KEY NOT NULL UNIQUE,
nome VARCHAR(100) NOT NULL,
idade INT NOT NULL CHECK (idade>=16),
email VARCHAR(150) NOT NULL UNIQUE,
senha VARCHAR(100) NOT NULL,
telefone VARCHAR(28) NOT NULL UNIQUE,
endereco VARCHAR(150) NOT NULL
);

CREATE TABLE usuario_log(
id_log INT AUTO_INCREMENT PRIMARY KEY,
cpf_cnpj_new VARCHAR(20),
cpf_cnpj_old VARCHAR(20),
nome_new VARCHAR(100),
nome_old VARCHAR(100),
idade_new INT,
idade_old INT,
email_new VARCHAR(150),
email_old VARCHAR(150),
senha_new VARCHAR(100),
senha_old VARCHAR(100),
telefone_new VARCHAR(28),
telefone_old VARCHAR(28),
endereco_new VARCHAR(150),
endereco_old VARCHAR(150)
);

CREATE TABLE plano(
id_plano INT PRIMARY KEY,
cpf_cnpj VARCHAR(20) NOT NULL UNIQUE,
tipo_plano VARCHAR(50) NOT NULL,
data_validade DATE NOT NULL,
data_criacao DATE NOT NULL
);

CREATE TABLE pagamento(
registro_pagamento INT PRIMARY KEY,
codigo_entregador INT NOT NULL UNIQUE,
cpf_cnpj VARCHAR(20) NOT NULL UNIQUE,
codSeguranca VARCHAR(100) NOT NULL,
numero_do_cartao VARCHAR(16) NOT NULL,
validade_do_cartao DATE NOT NULL,
numero_conta VARCHAR(30) NOT NULL
);

CREATE TABLE entrega(
id_entrega INT PRIMARY KEY,
codigo_entregador INT NOT NULL UNIQUE,
forma_entrega VARCHAR(20) NOT NULL,
taxa_entrega DECIMAL(5,2) NOT NULL,
codigo_confirmacao INT NOT NULL UNIQUE
);

CREATE TABLE anuncio(
id_anuncio INT PRIMARY KEY,
data_postagem DATE NOT NULL,
tipo_anuncio VARCHAR(100) NOT NULL,
valor DECIMAL(7,2) NOT NULL,
localizacao_anuncio VARCHAR(150) NOT NULL,
desconto_plano DECIMAL(7,2),
cpf_cnpj VARCHAR(20) NOT NULL UNIQUE
);

IN p_email VARCHAR(150),
IN p_senha VARCHAR(100)
)
BEGIN
UPDATE usuario
SET email = p_email,
senha = p_senha
WHERE cpf_cnpj = "00000000000";
END //

IN p_tipo_plano VARCHAR(50),
IN p_cpf_cnpj VARCHAR(20)
)
BEGIN
UPDATE plano
SET tipo_plano = p_tipo_plano
WHERE cpf_cnpj = p_cpf_cnpj;
END //

CREATE VIEW cons_verificar_transporte AS
SELECT entrega.codigo_entregador, entrega.forma_entrega, pagamento.registro_pagamento
FROM entrega
LEFT JOIN pagamento
ON entrega.codigo_entregador = pagamento.codigo_entregador
WHERE entrega.forma_entrega = 'carro';



CREATE VIEW cons_verificar_transporte_nulo AS
SELECT entrega.codigo_entregador, entrega.forma_entrega
FROM entrega
LEFT JOIN pagamento
ON entrega.codigo_entregador = pagamento.codigo_entregador
WHERE entrega.forma_entrega = 'carro'
AND pagamento.codigo_entregador IS NULL;



CREATE OR REPLACE VIEW vw_03_self AS
SELECT a1.id_anuncio AS anuncio_1,
       a2.id_anuncio AS anuncio_2,
       a1.localizacao_anuncio AS cidade
FROM anuncio a1
JOIN anuncio a2
  ON a1.localizacao_anuncio = a2.localizacao_anuncio
 AND a1.id_anuncio < a2.id_anuncio;



CREATE OR REPLACE VIEW vw_04_full AS
SELECT a.id_anuncio, a.tipo_anuncio, e.id_entrega, e.forma_entrega
FROM anuncio a
LEFT JOIN entrega e ON e.id_entrega = a.id_anuncio
UNION
SELECT a.id_anuncio, a.tipo_anuncio, e.id_entrega, e.forma_entrega
FROM anuncio a
RIGHT JOIN entrega e ON e.id_entrega = a.id_anuncio;




CREATE OR REPLACE VIEW vw_05_having AS
SELECT localizacao_anuncio AS cidade,
       COUNT(*) AS total_anuncios,
       SUM(valor) AS soma_valor
FROM anuncio
GROUP BY localizacao_anuncio
HAVING COUNT(*) >= 2;





CREATE OR REPLACE VIEW vw_06_top3 AS
SELECT id_anuncio, tipo_anuncio, valor
FROM anuncio
ORDER BY valor DESC
LIMIT 3;



CREATE OR REPLACE VIEW vw_07_subconsulta AS
SELECT id_anuncio, tipo_anuncio, valor
FROM anuncio
WHERE valor > (SELECT AVG(valor) FROM anuncio);






CREATE OR REPLACE VIEW vw_08_cte AS
WITH total_usuario AS (
    SELECT cpf_cnpj, SUM(valor) AS total_anuncios
    FROM anuncio
    GROUP BY cpf_cnpj
)
SELECT u.nome, t.total_anuncios
FROM total_usuario t
JOIN usuario u ON u.cpf_cnpj = t.cpf_cnpj;





CREATE OR REPLACE VIEW vw_09_ranking AS
SELECT id_plano,
       cpf_cnpj,
       tipo_plano,
       data_criacao,
       RANK() OVER (PARTITION BY YEAR(data_criacao) ORDER BY data_criacao DESC) AS posicao
FROM plano;




CREATE OR REPLACE VIEW vw_10_vizinho AS
SELECT id_anuncio,
       data_postagem,
       valor,
       LAG(valor) OVER (ORDER BY data_postagem, id_anuncio) AS valor_anterior,
       SUM(valor) OVER (ORDER BY data_postagem, id_anuncio) AS valor_acumulado
FROM anuncio;

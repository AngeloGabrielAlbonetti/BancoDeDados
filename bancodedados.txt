CREATE TABLE usuario( 
    cpf_cnpj VARCHAR(20) PRIMARY KEY NOT NULL UNIQUE, 
    nome VARCHAR(100) NOT NULL, 
    idade INT NOT NULL CHECK (idade>=16), 
    email VARCHAR(150) NOT NULL UNIQUE, 
    senha VARCHAR(100) NOT NULL, 
    telefone VARCHAR(28) NOT NULL UNIQUE, 
    endereco VARCHAR(150) NOT NULL 
); 

CREATE TABLE usuario_log( 
    id_log INT AUTO_INCREMENT PRIMARY KEY,
    cpf_cnpj_new VARCHAR(20), 
    cpf_cnpj_old VARCHAR(20), 
    nome_new VARCHAR(100), 
    nome_old VARCHAR(100), 
    idade_new INT, 
    idade_old INT, 
    email_new VARCHAR(150),
    email_old VARCHAR(150),  
    senha_new VARCHAR(100),
    senha_old VARCHAR(100), 
    telefone_new VARCHAR(28),
    telefone_old VARCHAR(28), 
    endereco_new VARCHAR(150),
    endereco_old VARCHAR(150)
); 


CREATE TABLE plano( 
    id_plano INT PRIMARY KEY, 
    cpf_cnpj VARCHAR(20) NOT NULL UNIQUE, 
    tipo_plano VARCHAR(50) NOT NULL UNIQUE, 
    data_validade DATE NOT NULL, 
    data_criacao DATE NOT NULL 
); 

CREATE TABLE pagamento( 
    registro_pagamento INT PRIMARY KEY, 
    codigo_entregador INT NOT NULL UNIQUE, 
    cpf_cnpj VARCHAR(20) NOT NULL UNIQUE, 
    codSeguranca VARCHAR(100) NOT NULL, 
    numero_do_cartao VARCHAR(16) NOT NULL, 
    validade_do_cartao DATE NOT NULL, 
    numero_conta VARCHAR(30) NOT NULL 
); 

CREATE TABLE entrega( 
    id_entrega INT PRIMARY KEY, 
    codigo_entregador INT NOT NULL UNIQUE, 
    forma_entrega VARCHAR(20) NOT NULL, 
    taxa_entrega DECIMAL(5,2) NOT NULL, 
    codigo_confirmacao INT NOT NULL UNIQUE 
); 

CREATE TABLE anuncio( 
    id_anuncio INT PRIMARY KEY, 
    data_postagem DATE NOT NULL, 
    tipo_anuncio VARCHAR(100) NOT NULL, 
    valor DECIMAL(7,2) NOT NULL, 
    localizacao_anuncio VARCHAR(150) NOT NULL, 
    desconto_plano DECIMAL(7,2), 
    cpf_cnpj VARCHAR(20) NOT NULL UNIQUE 
);

ALTER TABLE plano ADD CONSTRAINT fk_plano_anuncio FOREIGN KEY (cpf_cnpj) REFERENCES usuario(cpf_cnpj); 
ALTER TABLE anuncio ADD CONSTRAINT fk_anuncio_plano FOREIGN KEY (cpf_cnpj) REFERENCES plano(cpf_cnpj); 
ALTER TABLE pagamento ADD CONSTRAINT fk_pagamento_entrega FOREIGN KEY (codigo_entregador) REFERENCES entrega(codigo_entregador); 
ALTER TABLE pagamento ADD CONSTRAINT fk_pagamento_anuncio FOREIGN KEY (cpf_cnpj) REFERENCES anuncio(cpf_cnpj);

CREATE TRIGGER trg_usuario_log_update_after
AFTER UPDATE ON usuario_log
FOR EACH ROW 
BEGIN 
    INSERT INTO usuario_log (id_log   , cpf_cnpj_new, cpf_cnpj_old,     nome_new,     nome_old,    idade_new,    idade_old, email_new, 
                             email_old,    senha_new,    senha_old, telefone_new, telefone_old, endereco_new, endereco_old
    ) 
    VALUES (OLD.id_log, NEW.cpf_cnpj_new, 
        OLD.cpf_cnpj_old, 
        NEW.nome_new, 
        OLD.nome_old, 
        NEW.idade_new, 
        OLD.idade_old, 
        NEW.email_new, 
        OLD.email_old,  
        NEW.senha_new, 
        OLD.senha_old, 
        NEW.telefone_new, 
        OLD.telefone_old, 
        NEW.endereco_new, 
        OLD.endereco_old
	)
END;
